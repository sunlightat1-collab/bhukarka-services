var ss = SpreadsheetApp.getActiveSpreadsheet(); function doGet(e) { var action = e.parameter ? e.parameter.action : ""; if (action == "getData") { return getData(); } else if (action == "updateSettings") { return updateSettings(e.parameter.notice, e.parameter.status); } else if (action == "addService") { return addService(e.parameter); } else if (action == "deleteService") { return deleteService(e.parameter.title); } return ContentService.createTextOutput(JSON.stringify({"status": "success", "message": "API is running"})).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); if (data.action === "addServiceWithFile") { return addServiceWithFile(data); } // सामान्य रिक्वेस्ट सबमिशन के लिए var sheet = ss.getSheetByName("Requests"); if (!sheet) { sheet = ss.insertSheet("Requests"); sheet.appendRow(["Timestamp", "Name", "Mobile", "Service", "Fees", "Details", "Status"]); } sheet.appendRow([ new Date(), data.name || "", data.mobile || "", data.service || "", data.fees || "", data.details || "", "Pending" ]); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()})).setMimeType(ContentService.MimeType.JSON); } } function getData() { var servicesSheet = ss.getSheetByName("Services"); var settingsSheet = ss.getSheetByName("Settings"); if (!servicesSheet) { servicesSheet = ss.insertSheet("Services"); servicesSheet.appendRow(["Emoji", "Title", "ServiceName", "Fees", "Link", "Target", "Note", "InputField", "FileUrl"]); } if (!settingsSheet) { settingsSheet = ss.insertSheet("Settings"); settingsSheet.getRange(1, 1).setValue("BHUARKARKA SERVICES में आपका स्वागत है!"); settingsSheet.getRange(1, 2).setValue("ONLINE"); } var servicesData = []; var sRows = servicesSheet.getDataRange().getValues(); for (var i = 1; i < sRows.length; i++) { servicesData.push({ emoji: sRows[i][0], title: sRows[i][1], serviceName: sRows[i][2], fees: sRows[i][3], link: sRows[i][4], target: sRows[i][5], note: sRows[i][6], inputField: sRows[i][7], fileUrl: sRows[i][8] || "" }); } var notice = settingsSheet.getRange(1, 1).getValue() || "स्वागत है!"; var shopStatus = settingsSheet.getRange(1, 2).getValue() || "ONLINE"; var result = { services: servicesData, notice: notice, status: shopStatus }; return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } function updateSettings(notice, status) { var sheet = ss.getSheetByName("Settings"); if (!sheet) sheet = ss.insertSheet("Settings"); if(notice) sheet.getRange(1, 1).setValue(notice); if(status) sheet.getRange(1, 2).setValue(status); return ContentService.createTextOutput(JSON.stringify({"status": "updated"})).setMimeType(ContentService.MimeType.JSON); } function addService(params) { var sheet = ss.getSheetByName("Services"); if (!sheet) { sheet = ss.insertSheet("Services"); sheet.appendRow(["Emoji", "Title", "ServiceName", "Fees", "Link", "Target", "Note", "InputField", "FileUrl"]); } sheet.appendRow([ params.emoji || "", params.title || "", params.serviceName || "", params.fees || "", params.link || "", params.target || "", params.note || "", params.inputField || "", "" ]); return ContentService.createTextOutput(JSON.stringify({"status": "added"})).setMimeType(ContentService.MimeType.JSON); } function addServiceWithFile(data) { var sheet = ss.getSheetByName("Services"); if (!sheet) { sheet = ss.insertSheet("Services"); sheet.appendRow(["Emoji", "Title", "ServiceName", "Fees", "Link", "Target", "Note", "InputField", "FileUrl"]); } // Base64 फाइल को Google Drive में सेव करना var decodedBytes = Utilities.base64Decode(data.fileData); var blob = Utilities.newBlob(decodedBytes, '', data.fileName); // ड्राइव फोल्डर (आप चाहें तो अपनी पसंद का फोल्डर आईडी दे सकते हैं, फिलहाल रूट में सेव होगा) var file = DriveApp.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); var fileUrl = file.getUrl(); sheet.appendRow([ data.emoji || "", data.title || "", "", data.fees || "", data.link || "", "", "", "", fileUrl ]); return ContentService.createTextOutput(JSON.stringify({"status": "added_with_file", "fileUrl": fileUrl})).setMimeType(ContentService.MimeType.JSON); } function deleteService(title) { var sheet = ss.getSheetByName("Services"); if (!sheet) return ContentService.createTextOutput(JSON.stringify({"status": "not found"})).setMimeType(ContentService.MimeType.JSON); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][1] == title) { sheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput(JSON.stringify({"status": "deleted"})).setMimeType(ContentService.MimeType.JSON); }
